#include<bits/stdc++.h>
using namespace std;
int f(int n){
    int s=sqrt(n);
    if(n==2)
        return 3;
    if(s*s==n)
        return s+s;
    int mn=INT_MAX;
        mn=2*(s+1);
        for(int i=1;i<=s+1;i++){
           for(int j=1;j<i;j++){
                if(i*j>=n && (i+j)<mn){
                    mn=i+j;
                }
           }
        }
        return mn;
   }
int main(){
       int n;
   while(cin>>n){
     cout<<f(n)<<endl;
   }
        }
