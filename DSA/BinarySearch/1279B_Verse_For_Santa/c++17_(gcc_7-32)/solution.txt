#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--){
        int n,s;
        cin>>n>>s;
        int a[n+3];
        int mx=0,p=0;
        long long x=0;
        for(int i=0;i<n;i++){
            cin>>a[i];
            x+=a[i];
        }
        long long sum=0;
        int l=-1;
        bool flag=false;
        for(int i=0;i<n;i++){
            if(flag)
                break;
            sum+=a[i];
            if(a[i]>mx){
                mx=a[i];
                p=i;
            }
            if(sum>s){
                if(l==-1){
                    l=i;
                }
                if(sum-mx<=s){
                    if(i>=l){
                        flag=true;
                        break;
                    }
                }
                //cout<<sum<<" "<<s<<" "<<sum-mx<<" "<<i<<" "<<l<<endl;;
            }else if(sum==s){
                if(l==-1){
                    l=i;
                }
                if(sum-mx<=s){
                    if(i>l){
                        flag=true;
                        break;
                    }
                }
                //cout<<sum<<" "<<s<<" "<<sum-mx<<" "<<i<<" "<<l<<endl;;
            }
        }
        if(x<=s){
            cout<<0<<endl;
        }
        else if(flag==true){
            cout<<p+1<<endl;
        }else{
            cout<<0<<endl;
        }
    }
}
