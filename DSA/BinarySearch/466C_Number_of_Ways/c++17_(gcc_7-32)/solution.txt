#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    int a[n+2];
    long long int s=0;
    for(int i=1;i<=n;i++){
        cin>>a[i];
        s+=a[i];
    }
    if(s%3){
        cout<<0<<endl;
        return 0;
    }
     long long int one_third=s/3,two_third=(s/3)*2,sum=0,ans=0,x=0;
    for(int i=1;i<n;i++){
        sum+=a[i];
        if(sum==two_third){
            ans+=x;
        }
        if(sum==one_third){
            x++;
        }
    }
    cout<<ans<<endl;
}
