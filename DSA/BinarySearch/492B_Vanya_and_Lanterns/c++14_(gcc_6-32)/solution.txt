#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n,l;
    //freopen("input.txt","r+",stdin);
    cin>>n>>l;
    int a[n+3];
    for(int i=0;i<n;i++)
    {
        cin>>a[i];
    }
    sort(a,a+n);
    int maxdis=0;
    for(int i=1;i<n;i++)
    {
        maxdis=max(maxdis,a[i]-a[i-1]);
    }
    double mxd=.5*(double)maxdis;
    double ans=(double)max(a[0],l-a[n-1]);
    ans=max(mxd,ans);
    printf("%.10lf\n",ans);
}
