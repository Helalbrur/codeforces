#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t,a,b,c;
    cin>>t;
    while(t--)
    {
        int n=0,s=0;
        cin>>a>>b>>c;
        n=min(b,c/2);
        s+=3*n;
         b=b-n;
        c=c-2*n;
        n=min(a,b/2);
        s+=3*n;
        cout<<s<<endl;
     }
}
