#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--){
        int a[4];
        cin>>a[0]>>a[1]>>a[2];
        sort(a,a+3);
        if(a[2]==a[1]){
            if(a[0]<a[1]-1){
                a[0]++;
                a[1]--;
                a[2]--;
             }else if(a[0]<a[1]){
                a[0]++;
            }
         }else if(a[0]==a[1]){
            if(a[2]>a[1]+1){
                a[2]--;
                a[1]++;
                a[0]++;
            }else if(a[2]>a[1]){
                a[2]--;
            }
        }else{
            if(a[0]<a[1]){
                a[0]++;
            }
            if(a[2]>a[1]){
                a[2]--;
            }
         }
         cout<<(long long)abs(a[0]-a[1])+(long long)abs(a[0]-a[2])+(long long)abs(a[2]-a[1])<<endl;
    }
}
