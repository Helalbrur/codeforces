#include<bits/stdc++.h>
using namespace std;
int main()
{
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    int n,m;
    cin>>n>>m;
    int a[n+2];
    for(int i=1;i<=n;i++){
        cin>>a[i];
    }
    int s=1;
    if(n>m){
        cout<<0<<"\n";
        return 0;
    }
    for(int i=1;i<=n;i++){
        for(int j=i+1;j<=n;j++){
            s=((s%m)*((max(a[i],a[j])-min(a[i],a[j]))%m))%m;
        }
    }
    cout<<s<<"\n";
}
