#include<bits/stdc++.h>
using namespace std;
int main()
{
    long int a,b,c,d;
    cin>>a>>b;
    cin>>c>>d;
    map<long int,long int> mp;
        mp[b]++;
        mp[d]++;
    for(long int i=1;i<=10000;i++)
    {
        long int f=a*i;
        long int s=c*i;
        mp[f+b]++;
        mp[s+d]++;
    }
    for(auto it=mp.begin();it!=mp.end();++it)
    {
        if(it->second>1)
        {
            cout<<it->first<<endl;
            return 0;
        }
        //cout<<it->first<<" "<<it->second<<endl;
    }
    cout<<"-1"<<endl;
}
