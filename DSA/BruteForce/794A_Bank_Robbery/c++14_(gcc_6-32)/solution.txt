#include<bits/stdc++.h>

int main()
{
    ios_base::sync_with_stdio(0);
    cin.tie(NULL);
    int a,b,c;
    //freopen("input.txt","r",stdin);
    cin>>a>>b>>c;
    int n,x,cnt=0;
    cin>>n;
    for(int i=0;i<n;i++)
    {
        cin>>x;
        if( x>b && x<c)
            cnt++;
    }
    cout<<cnt<<endl;
}
