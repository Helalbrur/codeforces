#include<bits/stdc++.h>
using namespace std;
int main()
{
    string s;
    cin>>s;
    int l=s.length();
    int cnt=0;
    int x;
    if(l==1)
    {
       cout<<"YES"<<endl;
       return 0;
    }
    if( l%2==0)
    {
        x=l/2-1;
    }
    else
        x=l/2;
    for(int i=0;i<=x;i++)
    {
        if(s[i]!=s[l-i-1])
            cnt++;
    }
    //cout<<x<<endl;
    if(l%2==1)
    {
        if(cnt==0)
        {
            cout<<"YES"<<endl;
            return 0;
        }
    }
    if(cnt!=1)
    {
        cout<<"NO"<<endl;
    }
    else
    {
        cout<<"YES"<<endl;
    }
}
