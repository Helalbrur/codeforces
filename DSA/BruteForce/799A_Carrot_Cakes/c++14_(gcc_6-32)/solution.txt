#include<bits/stdc++.h>
using namespace std;
int main()
{
    ios_base::sync_with_stdio(0);
    cin.tie(NULL);
    int n,t,k,d;
    cin>>n>>t>>k>>d;
     if(k>=n)
    {
        cout<<"NO"<<endl;
        return 0;
    }
    else if(t>d)
    {
        cout<<"YES"<<endl;
        return 0;
    }
    int r=(n-k)%k;
    if(r)
        r=1;
    if((((n-k)/k +r)*t)>d)
    {
        cout<<"YES"<<endl;
        return 0;
    }
    cout<<"NO"<<endl;
 }
