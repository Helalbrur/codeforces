#include<bits/stdc++.h>
using namespace std;
int main(){
        int n,k;
    cin>>n>>k;
    if(k==1||k==n){
        cout<<n*3<<endl;
    }else{
        cout<<n*3+min(k-1,n-k)<<endl;
    }
}
