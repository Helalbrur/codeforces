#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    int a[n+2];
    vector<int> x,y;
    map<int,int> mp;
    for(int i=0;i<n;i++){
        cin>>a[i];
        mp[a[i]]++;
        if(mp[a[i]]==1){
            x.push_back(a[i]);
        }else{
            y.push_back(a[i]);
        }
    }
    for(auto it=mp.begin();it!=mp.end();++it){
        if(it->second>2){
            cout<<"NO"<<endl;
            return 0;
        }
    }
    cout<<"YES"<<endl;
    sort(x.begin(),x.end());
    sort(y.begin(),y.end(),greater<int>());
    cout<<x.size()<<endl;
    for(int i=0;i<x.size();i++){
        if(i>0){
            cout<<" ";
        }
        cout<<x[i];
    }
    cout<<endl<<y.size()<<endl;
    for(int i=0;i<y.size();i++){
        if(i>0){
            cout<<" ";
        }
        cout<<y[i];
    }
}
