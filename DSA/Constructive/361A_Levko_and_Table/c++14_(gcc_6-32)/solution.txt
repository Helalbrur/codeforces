#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n,k;
    cin>>n>>k;
    for(int i=1;i<=n;i++)
    {
        for(int j=1;j<=n;j++)
        {
            if(j>1)
                cout<<" ";
            if((i==j))
            {
                cout<<k-n+1;
            }
            else
            {
                cout<<1;
            }
        }
        cout<<endl;
    }
}
