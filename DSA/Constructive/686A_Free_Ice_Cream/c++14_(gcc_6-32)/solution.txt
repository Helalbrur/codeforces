#include<iostream>
using namespace std;
int main()
{
    int64_t n,x;
    cin>>n>>x;
    int64_t cnt=0;
    for(int i=0;i<n;i++){
        char c;
        int64_t p;
        cin>>c>>p;
        if(c=='+'){
            x+=p;
        }else{
            if(x>=p){
                x-=p;
            }else{
                cnt++;
            }
        }
    }
    cout<<x<<" "<<cnt<<endl;
}
