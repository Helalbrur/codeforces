#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n,x=0,y=0;
    cin>>n;
    for(int i=0;i<n;i++)
    {
        if(x<2)
        {
            cout<<"a";
            x++;
            y=0;
        }
        else if(y<2)
        {
            cout<<"b";
            y++;
        }
        if(y==2)
            x=0;
     }
}
