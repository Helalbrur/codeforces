#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n,t;
    cin>>t;
    while(t--){
    cin>>n;
    int a[n+3];
    int b[n+3];
       int mn=INT_MAX;
    for(int i=0;i<n;i++){
        cin>>a[i];
     }
    for(int i=n-1;i>=0;i--){
        mn=min(mn,a[i]);
        b[i]=mn;
    }
      long long cnt=0;
    for(int i=0;i<n;i++){
        if(a[i]>b[i]){
             cnt++;
        }
    }
    cout<<cnt<<endl;
  }
}
