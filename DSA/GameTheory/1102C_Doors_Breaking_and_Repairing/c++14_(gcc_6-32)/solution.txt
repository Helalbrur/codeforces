#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,x,y;
    cin>>n>>x>>y;
    int a[n+5];
    int x_c=0;
    for(int i=0;i<n;i++){
        cin>>a[i];
        if(a[i]<=x){
            x_c++;
        }
    }
    if(x>y){
        cout<<n<<endl;
            }
    else if(x_c%2==0){
        cout<<x_c/2<<endl;
            }
    else{
        cout<<(x_c/2)+1<<endl;
            }
}
