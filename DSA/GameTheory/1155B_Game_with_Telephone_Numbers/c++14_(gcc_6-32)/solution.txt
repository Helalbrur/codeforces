#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    string input;
    cin>>input;
    int p=n-11,cnt=0;
    for(int i=0;i<=p && i<n;i++){
        if(input[i]=='8'){
            cnt++;
        }
    }
    int x;
    if(p%2==0){
        x=p/2;
    }else{
        x=(p-1)/2;
    }
    //cout<<p<<" "<<cnt<< " "<<x<<endl;
    if(cnt>x){
        cout<<"YES"<<endl;
    }else{
        cout<<"NO"<<endl;
    }
}
