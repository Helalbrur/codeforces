#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--){
        int n,k1,k2;
        cin>>n>>k1>>k2;
        int mx_f=0,mx_s=0;
        for(int i=0;i<k1;i++){
            int x;
            cin>>x;
            mx_f=max(mx_f,x);
        }
        for(int i=0;i<k2;i++){
            int x;
            cin>>x;
            mx_s=max(mx_s,x);
        }
        if(mx_f>mx_s){
            cout<<"YES"<<endl;
        }else{
            cout<<"NO"<<endl;
        }
    }
}
