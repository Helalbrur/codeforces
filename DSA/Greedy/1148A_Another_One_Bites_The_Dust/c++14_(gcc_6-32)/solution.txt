#include<iostream>
using namespace std;
int main(){
    int a,b,c;
    cin>>a>>b>>c;
    int64_t cnt=(int64_t)2*c+(int64_t)2*min(a,b);
    if(max(a,b)>min(a,b)){
        cnt++;
    }
    cout<<cnt<<endl;
}
