int main()
{
    int n;
    cin>>n;
    int a[n+2];
    deque<int> d;
    for(int i=1;i<=n;i++){
        cin>>a[i];
        d.push_back(a[i]);
    }
        string s;
    int last=INT_MIN;
    while(!d.empty()){
        int f=d.front();
        int b=d.back();
        if(b>last && f>last){
            if(b<f){
                s.push_back('R');
                d.pop_back(); 
                last=b;
            }else{
                s.push_back('L');
                d.pop_front();
                last=f;
            }
                   }else if(b>last){
                        s.push_back('R');
             d.pop_back(); 
             last=b;
        }else if(f>last){
             s.push_back('L');
            d.pop_front();
            last=f;
        }else{
            break;
        }
    }
    cout<<s.size()<<endl;
    cout<<s<<endl;
}
