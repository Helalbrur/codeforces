#include<bits/stdc++.h>
using namespace std;
typedef long long int ll;
int main()
{
    ll n,m;
    //freopen("1.txt","r+",stdin);
    ll fn=1000000010,fm=0;
    cin>>n;
    for(ll i=0;i<n;i++)
    {
        ll a,b;
        cin>>a>>b;
        fn=min(fn,b);
        fm=max(fm,a);
    }
    cin>>m;
    ll sm=0,sn=1000000010;
    for(ll i=0;i<m;i++)
    {
        ll a,b;
        cin>>a>>b;
        sm=max(sm,a);
        sn=min(sn,b);
    }
    if(sm>fn && fm>sn)
    {
         //cout<<"f"<<endl;
        if((sm-fn)>(fm-sn))
        {
            cout<<sm-fn<<endl;
        }
        else
           cout<<fm-sn<<endl;
    }
    else if(sm>fn)
        cout<<sm-fn<<endl;
    else if(fm>sn)
        cout<<fm-sn<<endl;
    else
        cout<<0<<endl;
}
