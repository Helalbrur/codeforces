#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int main()
{
    ios_base::sync_with_stdio(0);
    cin.tie(NULL);
    ll n,f;
    //freopen("input.txt","r",stdin);
    cin>>n>>f;
    ll a,b,cnt=0;
    priority_queue<ll> q;
    for(ll i=0;i<n;i++)
    {
        cin>>a>>b;
        ll x=min(a,b);
        ll y=min(a*2,b);
        q.push(y-x);
        cnt+=x;
    }
     for(ll i=1;i<=f;i++)
    {
        cnt+=q.top();
        q.pop();
    }
    cout<<cnt<<endl;
}
