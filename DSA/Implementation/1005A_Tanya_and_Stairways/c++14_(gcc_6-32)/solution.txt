#include<iostream>
#include<string.h>
using namespace std;
int main(){
    int n;
    cin>>n;
    int a[n+2],b[n+2];
    memset(b,0,sizeof(b));
    int c=1;
    for(int i=1;i<=n;i++){
        cin>>a[i];
        if(i>1){
            if(a[i]<=a[i-1]){
                c++;
            }
         }
        b[c]++;
     }
    cout<<c<<endl;
    for(int i=1;i<=c;i++){
        cout<<b[i]<<" ";
    }
    cout<<endl;
  }
