#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    string a,b;
    cin>>a;
    int i=0,len=a.length(),cnt=0;
    while(cnt<len){
        cnt+=(i+1);
        b+=a[cnt-1];
        i++;
    }
    cout<<b<<endl;
}
