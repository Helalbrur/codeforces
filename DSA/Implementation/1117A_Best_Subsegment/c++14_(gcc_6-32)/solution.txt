#include<bits/stdc++.h>
using namespace std;
int main()
{
 int n,m,k;
 cin>>n;
 int a[n+2];
 int mx=0,cnt=1,f=0;
 for(int i=0; i<n;i++){
    cin>>a[i];
    mx=max(mx,a[i]);
  }
bool flag=0;
 for(int i=0; i<n;i++){
    if(a[i]==mx ){
        if(flag==1){
            cnt++;
        }
        flag=1;
        f=max(cnt,f);
    }else{
        flag=0;
        cnt=1;
        f=max(cnt,f);
    }
  }
cout<<f<<endl;
  }
