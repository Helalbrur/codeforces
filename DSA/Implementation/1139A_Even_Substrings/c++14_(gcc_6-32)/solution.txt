#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin>>n;
    string text;
    cin>>text;
    int cnt=0;
    for(int i=0;i<n;i++){
        if((text[i]-'0')%2==0){
            cnt+=i+1;
        }
    }
    cout<<cnt<<endl;
}
