#include<bits/stdc++.h>
using namespace std;
int main(){
        int n,m,r=-1,cnt=0;
    cin>>n>>m;
    if(m%n==0){
        r=0;
       int d=m/n;
        while(d%2==0){
            d/=2;
            cnt++;
        }
        while(d%3==0){
            d/=3;
            cnt++;
        }
        if(d!=1){
            r=-1;
                   }else{
            r=cnt;
        }
        cout<<r<<endl;
    }else{
        cout<<r<<endl;
    }
}
