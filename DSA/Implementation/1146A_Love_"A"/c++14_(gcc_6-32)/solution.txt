#include<bits/stdc++.h>
using namespace std;
int main()
{
    string text;
    cin>>text;
    int cnt=0,len=text.length();
    for(int i=0;i<len;i++){
        if(text[i]=='a'){
            cnt++;
        }
    }
    cout<<min(len,(cnt*2-1))<<endl;
    return 0;
}
