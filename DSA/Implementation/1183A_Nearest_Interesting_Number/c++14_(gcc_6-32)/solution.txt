#include<bits/stdc++.h>
using namespace std;
int get(int a){
        while(true){
        int n=a;
       int sum=0;
        while(n){
                            sum+=n%10;
            n=n/10;
        }
        if(sum%4==0){
            return a;
        }
        a++;
    }
}
int main(){
    int a;
    cin>>a;
    cout<<get(a)<<endl;
}
