#include<bits/stdc++.h>
using namespace std;
int main()
{
    int k;
    cin>>k;
    while(k--){
        int n;
        cin>>n;
        int len=1;
        int a[n+2];
        for(int i=0;i<n;i++){
          cin>>a[i];
        }
        sort(a,a+n);
         for(int i=0;i<n;i++){
            if(a[i]>=n-i){
                len=n-i;
                break;
            }
        }
        cout<<len<<endl;
    }
}
