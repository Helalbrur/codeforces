#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    int a[n+3];
    int64_t sum=0;
    a[0]=0;
    for(int i=1;i<=n;i++){
        cin>>a[i];
        sum+=a[i];
        int c=0;
        if(a[i-1]<0){
           c=-a[i-1];
        }
        sum+=c;
        cout<<sum<<" ";
    }
  }
