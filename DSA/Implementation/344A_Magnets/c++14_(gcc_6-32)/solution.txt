#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    string a,b;
    for(int i=0;i<n;i++){
        cin>>b;
        a+=b;
    }
    int cnt=0;
    for(int i=0;i<a.length()-1;i++){
        if(a[i]==a[i+1]){
            cnt++;
        }
    }
    cout<<cnt+1<<endl;
}
