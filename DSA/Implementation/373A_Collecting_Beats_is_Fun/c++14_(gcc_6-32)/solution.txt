#include<bits/stdc++.h>
using namespace std;
int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    //freopen("411b.txt","r",stdin);
    int n;
    cin>>n;
    string g[5];
    map<int,int> mp;
    for(int i=0;i<4;i++)
    {
        cin>>g[i];
    }
    for(int i=0;i<4;i++)
    {
        for(int j=0;j<4;j++)
        {
            if(isdigit(g[i][j]))
            {
                int x=g[i][j]-'0';
                mp[x]++;
                //cout<<x<<" "<<mp[x]<<endl;
            }
         }
        //cout<<endl<<endl;
    }
    for(auto it=mp.begin();it!=mp.end();++it)
    {
        if(it->second>2*n)
        {
            cout<<"NO"<<endl;
            return 0;
        }
    }
    cout<<"YES"<<endl;
}
