#include<bits/stdc++.h>
using namespace std;
int main()
{
    string text;
    //freopen("input.txt","r",stdin);
    while(cin>>text)
    {
        long long int mid=0;
        for(long long int i=0;i<text.length();i++)
        {
            if(text[i]=='^')
            {
                mid=i;
                break;
            }
         }
        long long int res=0;
        for(long long int i=0;i<text.length();i++)
        {
            if(isdigit(text[i]))
            {
                res+=((text[i]-'0')*(i-mid));
            }
         }
        if(res==0)
        {
            cout<<"balance"<<endl;
        }
        else if(res<0)
        {
            cout<<"left"<<endl;
        }
        else
        {
            cout<<"right"<<endl;
        }
    }
}
