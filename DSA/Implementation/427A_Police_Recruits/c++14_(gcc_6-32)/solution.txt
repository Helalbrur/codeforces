#include<iostream>
using namespace std;
int main()
{
    int n;
    cin>>n;
    int c=0;
    int cnt=0;
    while(n--){
        int a;
        cin>>a;
        c+=a;
        if(c<0){
            cnt++;
            c=0;
        }
    }
    cout<<cnt<<endl;
}
