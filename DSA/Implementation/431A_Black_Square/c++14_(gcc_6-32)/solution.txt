#include<iostream>
using namespace std;
int main(){
     int a[5];
    cin>>a[1]>>a[2]>>a[3]>>a[4];
    string b;
    cin>>b;
    int sum=0;
    for(int i=0;i<b.length();i++){
        sum+=a[b[i]-'0'];
    }
    cout<<sum<<endl;
}
