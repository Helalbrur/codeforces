#include<iostream>
#include<string>
#include<math.h>
using namespace std;
int main()
{
    int n,h;
    cin>>n>>h;
    int a,cnt=0;
    for(int i=1;i<=n;i++){
        cin>>a;
        if(a>h){
            cnt+=2;
        }else{
            cnt++;
        }
    }
    cout<<cnt<<endl;
}
