#include<iostream>
using namespace std;
int main()
{
    int n;
    cin>>n;
    int m,c,m_c=0,c_c=0;
    while(n--){
        cin>>m>>c;
        if(m>c){
            m_c++;
        }else if(c>m){
            c_c++;
        }
    }
    if(m_c>c_c){
        cout<<"Mishka"<<endl;
    }else if(c_c>m_c){
        cout<<"Chris"<<endl;
    }else{
        cout<<"Friendship is magic!^^"<<endl;
    }
  }
