#include<iostream>
using namespace std;
int main()
{
    int a,b,c;
    cin>>a>>b>>c;
    int p=1000;
    for(int i=min(a,min(b,c));i<=max(a,max(b,c));i++){
        p=min(p,abs(i-a)+abs(i-b)+abs(i-c));
    }
    cout<<p<<endl;
}
