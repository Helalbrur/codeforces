#include<iostream>
using namespace std;
int main()
{
    string b;
     cin>>b;
    int pos=26;
    int cnt=0;
    for(int i=0;i<b.length();i++){
        int j=b[i]-'a';
        cnt+=min(abs(j-pos),26-abs(j-pos));
        pos=j;
    }
    cout<<cnt<<endl;
}
