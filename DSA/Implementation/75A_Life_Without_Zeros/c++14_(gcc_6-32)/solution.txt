#include<bits/stdc++.h>
#include<string>
using namespace std;
int remz(int a){
   int n=0,ten=1;
    while(a)
    {
        int r=a%10;
        a=a/10;
         if(r)
        {
            n+=r*ten;
            ten*=10;
        }
    }
    return n;
}
int main()
{
    int a,b;
    cin>>a>>b;
     int c=a+b;
   if(remz(c)==remz(a)+remz(b))
   {
       cout<<"YES"<<endl;
   }
   else
   {
       cout<<"NO"<<endl;
   }
}
