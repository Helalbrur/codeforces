#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    //freopen("1.txt","r+",stdin);
    cin>>n;
    map<string,int> mp;
    mp["Tetrahedron"]=4;
    mp["Cube"]=6;
    mp["Octahedron"]=8;
    mp["Dodecahedron"]=12;
    mp["Icosahedron"]=20;
    int total=0;
    for(int i=0;i<n;i++)
    {
        string st;
        cin>>st;
        total+=mp[st];
    }
    cout<<total<<endl;
}
