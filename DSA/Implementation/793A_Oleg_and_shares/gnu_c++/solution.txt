#include<bits/stdc++.h>
using namespace std;
int main()
{
     ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    //freopen("411a.txt","r",stdin);
    long long int n,k,mn=1000000000,cnt=0;
    cin>>n>>k;
    long long int a[n+4];
    set<long long int> s;
     for(int i=1;i<=n;i++)
    {
        cin>>a[i];
        mn=min(a[i],mn);
        s.insert(a[i]%k);
        if(s.size()>1)
        {
           cout<<"-1"<<endl;
           return 0;
        }
     }
    for(int i=1;i<=n;i++)
    {
        cnt+=((a[i]-mn))/k;
    }
    cout<<cnt<<endl;
}
