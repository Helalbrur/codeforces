#include<bits/stdc++.h>
using namespace std;
int main()
{
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n,m,k;
    cin>>n>>m>>k;
    int b[n+3],a[n+3];
    memset(a,0,sizeof a);
    memset(b,0,sizeof b);
    for(int i=1;i<=n;i++)
        a[i]=i;
    for(int i=1;i<=m;i++)
    {
        int x;
        cin>>x;
        b[x]=1;
    }
    if(b[1]==1)
    {
        cout<<1<<endl;
        return 0;
    }
    bool flag=false;
    int pos=1;
    for(int i=1;i<=k;i++)
    {
        int x,y;
        cin>>x>>y;
        if(flag==false)
        {
            if(a[y]==1)
            {
                pos=x;
                if(b[x]==1)
                   flag=true;
            }
             if(a[x]==1)
            {
                pos=y;
                if(b[y]==1)
                    flag=true;
            }
            swap(a[x],a[y]);
        }
    }
    cout<<pos<<endl;
}
