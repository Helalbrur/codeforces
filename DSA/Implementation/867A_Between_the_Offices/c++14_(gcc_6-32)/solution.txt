#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin>>n;
    string a;
    cin>>a;
    int f=0,s=0;
    for(int i=0;i<n;i++){
        if(a[i]=='F'){
            if(i+1<n && a[i+1]=='S'){
                s++;
            }
        }
        else{
           if(i+1<n && a[i+1]=='F'){
                f++;
           }
        }
    }
     if(f>s){
        cout<<"YES"<<endl;
    }
    else{
        cout<<"NO"<<endl;
    }
 }
