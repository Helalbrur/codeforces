#include<iostream>
#include<string.h>
using namespace std;
int main()
{
    int n,m;
    cin>>n>>m;
    string a;
    cin>>a;
    for(int i=1;i<=m;i++){
        int l,r;
        char c,d;
        cin>>l>>r>>c>>d;
        l--;
        r--;
        for(int j=l;j<=r;j++){
            if(a[j]==c){
                a[j]=d;
            }
        }
    }
    cout<<a<<endl;
}
