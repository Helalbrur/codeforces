#include<bits/stdc++.h>
using namespace std;
int main(){
        int n,k,red,green,blue,cnt;
    cin>>n>>k;
     red=n*2;
     green=n*5;
     blue=n*8;
     cnt=0;
    if(red%k==0){
        cnt+=red/k;
            }else{
        cnt+=red/k+1;
    }
    if(green%k==0){
        cnt+=green/k;
            }else{
        cnt+=green/k+1;
    }
    if(blue%k==0){
        cnt+=blue/k;
            }else{
        cnt+=blue/k+1;
    }
    cout<<cnt<<endl;
 }
