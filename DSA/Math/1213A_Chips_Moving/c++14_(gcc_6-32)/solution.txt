#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    int a[n+3];
    long long sum=0;
    for(int i=0;i<n;i++){
        cin>>a[i];
        sum+=a[i];
    }
    sort(a,a+n);
     long long int mx=LONG_MAX;
    for(int i=0;i<n;i++){
        long long cnt=0;
        for(int j=0;j<n;j++){
            int x=abs(a[i]-a[j]);
             if(x%2==1){
                cnt+=1;
            }
        }
         mx=min(cnt,mx);
         cnt=0;
    }
    cout<<mx<<endl;
}
