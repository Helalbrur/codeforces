#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    while(n--){
       int64_t c,s;
        cin>>c>>s;
        if(c>=s){
            cout<<s<<endl;
        }
        else if(s%c==0){
            cout<<c*(s/c)*(s/c)<<endl;
        }else{
            int64_t p=s%c;
            int64_t x=s/c;
            cout<<p*(x+1)*(x+1)+(c-p)*(x*x)<<endl;
        }
    }
}
