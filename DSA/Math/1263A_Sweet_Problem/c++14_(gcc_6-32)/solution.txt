#include<bits/stdc++.h>
using namespace std;
#define line "\n"
int main(){
    int n;
    cin>>n;
    while(n--){
        int a[5];
        cin>>a[0]>>a[1]>>a[2];
        sort(a,a+3);
        if(a[0]+a[1]>=a[2]){
            cout<<a[2]+(a[0]+a[1]-a[2])/2<<endl;
        }else{
              cout<<min(a[0]+a[1],a[2])<<endl;
        }
     }
}
