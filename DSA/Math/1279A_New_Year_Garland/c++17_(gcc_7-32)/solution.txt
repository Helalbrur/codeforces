#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--){
        int f[4];
        cin>>f[0]>>f[1]>>f[2];
        sort(f,f+3);
        if(f[2]>f[1]+f[0]+1){
            cout<<"No"<<endl;
        }else{
            cout<<"Yes"<<endl;
        }
    }
}
