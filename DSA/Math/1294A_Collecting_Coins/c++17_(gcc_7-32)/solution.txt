#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--){
        long long int a,b,c,n;
        cin>>a>>b>>c>>n;
        long long int s=a+b+c;
        long long int mx=max(a,max(b,c));
        long long int r=mx*3-s;
        if(n>=r && (n-r)%3==0){
            cout<<"YES"<<endl;
        }else{
            cout<<"NO"<<endl;
        }
    }
}
