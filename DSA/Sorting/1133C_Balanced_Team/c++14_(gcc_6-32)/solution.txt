#include<bits/stdc++.h>
using namespace std;
int main(){
    //cout<<1<<endl;
    int n;
    cin>>n;
    int a[n+2];
    int mx=0;
    set<int> s;
    map<int,int> mp;
    //vector<int> vec;
    for(int i=0;i<n;i++){
        cin>>a[i];
        mp[a[i]]++;
        s.insert(a[i]);
        //vec.push_back(a[i]);
    }
   /* for(pair <int,int> it:mp){
        cout<<it.first<<" "<<it.second<<endl;
    }
    for(int it:vec){
        cout<<it<<endl;
    }
    */
    for(int i:s){
        int cur=mp[i];
        for(int j=1;j<=5;j++){
            cur+=mp[i+j];
        }
        mx=max(mx,cur);
    }
    cout<<mx<<endl;
}
