#include<bits/stdc++.h>
using namespace std;
int main()
{
   ios_base::sync_with_stdio(0);
   cin.tie(0);
   cout.tie(0);
   int k;
   cin>>k;
   while(k--){
    int n;
    cin>>n;
    string s,t;
    cin>>s;
    cin>>t;
    if(s!=t){
         int cnt=0;
        bool ok=false;
        vector<int> pos;
        for(int i=0;i<n;i++){
            if(s[i]!=t[i]){
                if(pos.size()>=3){
                    break;
                }
                pos.push_back(i);
             }
        }
         if(pos.size()==2){
           char a=s[pos[0]];
           char b=s[pos[1]];
           char c=t[pos[0]];
           char d=t[pos[1]];
           if(a==b && c==d){
                cout<<"Yes"<<endl;
            }else{
                cout<<"No"<<endl;
            }
        }else if(pos.size()==0){
            cout<<"Yes"<<endl;
        }else{
            cout<<"No"<<endl;
        }
     }else{
        cout<<"Yes"<<endl;
    }
    }
}
