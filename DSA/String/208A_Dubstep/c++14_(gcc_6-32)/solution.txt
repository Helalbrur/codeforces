#include <bits/stdc++.h>
 using namespace std;
 int main ()
{
    string a;
    int cnt=0, flag=0;
   // freopen("input.txt", "r", stdin);
    cin>>a;
    int l= a.size();
    bool fl=false;
    for(int i=0; i<l; i++)
    {
        if(a[i]=='W' && a[i+1]=='U'&& a[i+2]=='B')
        {
            cnt++;
            if(cnt==1 && fl)
                cout<<" ";
            i+=2;
        }
        else
        {
            cout<<a[i];
            fl=true;
            cnt=0;
        }
    }
    if(fl==false)
    {
        cout<<"WUB"<<endl;
    }
    return 0;
}
