#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    int a[n+2];
    map<int,int> mp;
    for(int i=0;i<n;i++){
        cin>>a[i];
        mp[a[i]]++;
    }
    int i=0;
    for(auto it=mp.begin();it!=mp.end();++it){
        i=max(it->second,i);
    }
    cout<<i<<endl;
}
