#include<iostream>
#include<algorithm>
using namespace std;
struct student{
    int id;
    int sum;
};
bool cmp(student a,student b){
     if(a.sum==b.sum){
        return a.id<b.id;
    }else{
        return a.sum>b.sum;
    }
}
int main()
{
    int n;
    cin>>n;
    student a[n+2];
    for(int i=1;i<=n;i++){
        int u,v,x,y;
        cin>>u>>v>>x>>y;
        a[i-1].id=i;
        a[i-1].sum=u+v+x+y;
    }
    sort(a,a+n,cmp);
    for(int i=0;i<n;i++){
        if(a[i].id==1){
            cout<<i+1<<endl;
            break;
        }
    }
 }
