#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,k;
    //freopen("input.txt","r+",stdin);
    cin>>n>>k;
    char text[n+4];
    int cnt[27];
    memset(cnt,0,sizeof(cnt));
    cin>>text;
    for(int i=0;i<n;i++){
        cnt[text[i]-'A']++;
    }
    int cmax=n;
    for(int i=0;i<k;i++){
        cmax=min(cmax,cnt[i]);
    }
    cout<<cmax*k<<endl;
     }
