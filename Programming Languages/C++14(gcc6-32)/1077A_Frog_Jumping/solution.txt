#include<iostream>
using namespace std;
int main()
{
    int q;
    cin>>q;
    while(q--){
        int64_t a,b,k;
        cin>>a>>b>>k;
        int64_t p,n=k/2;
        p=n;
        if(k%2){
            p++;
        }
        cout<<a*p-b*n<<endl;
    }
}
