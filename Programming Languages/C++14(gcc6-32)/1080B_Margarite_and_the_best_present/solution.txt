#include<bits/stdc++.h>
using namespace std;
long long f(long long x){
    if(x%2==0){
        return x/2;
    }
    return -x+f(x-1);
}
int main(){
        long long int q,l,r;
    cin>>q;
    while(q--){
        cin>>l>>r;
        cout<<f(r)-f(l-1)<<endl;
    }
}
