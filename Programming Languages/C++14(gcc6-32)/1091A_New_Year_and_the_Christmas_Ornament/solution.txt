#include<bits/stdc++.h>
using namespace std;
 int main(){
    int a[3];
    int mn=200;
    for(int i=0;i<3;i++){
        cin>>a[i];
        if(mn>a[i]){
            mn=a[i];
        }
    }
    if(mn==a[2]){
        cout<<mn*3-3<<endl;
    }
    else if(a[1]==mn){
        if(a[2]>a[1]){
            cout<<mn*3<<endl;
        }else{
            cout<<mn*3-3<<endl;
        }
    }
    else if(a[1]>mn && a[2]>mn+1){
        cout<<mn*3+3<<endl;
    }
    else if(a[2]>mn){
        cout<<mn*3<<endl;
    }else{
        cout<<mn*3-3<<endl;
    }
}
