#include<iostream>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        int cnt=0;
        int a=n/7;
        cnt+=a;
        n=n-a*7;
        if(n){
            cnt++;
        }
        cout<<cnt<<endl;
     }
}
