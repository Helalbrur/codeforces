#include<bits/stdc++.h>
using namespace std;
int n,flag=0;
int a[20];
void dp(int sum,int i){
    if(i==n){
        if(sum%360==0){
            flag=1;
        }
        return;
    }
    dp(sum+a[i],i+1);
    dp(sum-a[i],i+1);
}
int main(){
    cin>>n;
        for(int i=0;i<n;i++){
        cin>>a[i];
    }
    dp(0,0);
    if(flag==1){
        cout<<"YES"<<endl;
    }else{
        cout<<"NO"<<endl;
    }
}
