#include<bits/stdc++.h>
using namespace std;
 int main(){
       int h,w,u1,d1,u2,d2;
   cin>>w>>h;
   cin>>u1>>d1;
   cin>>u2>>d2;
   while(h){
                w+=h;
        if(h==d1){
            w=w-u1;
        }
        if(h==d2){
            w=w-u2;
        }
        if(w<0){
            w=0;
        }
        h--;
   }
   cout<<w<<endl;
}
