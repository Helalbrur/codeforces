#include<bits/stdc++.h>
using namespace std;
int main()
{
    int q;
    cin>>q;
    int  l1,r1,l2,r2;
    for(int i=1;i<=q;i++){
        cin>>l1>>r1>>l2>>r2;
        if(l1!=l2){
            cout<<l1<<" "<<l2<<endl;
        }else if(r1-l1>1){
            cout<<l1+1<<" "<<l2<<endl;
        }else if(r2-l2>1){
            cout<<l1<<" "<<l2+1<<endl;
        }else{
            cout<<l1<<" "<<r2<<endl;
        }
    }
}
