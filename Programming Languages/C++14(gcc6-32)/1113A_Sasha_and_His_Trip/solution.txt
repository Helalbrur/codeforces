#include<bits/stdc++.h>
using namespace std;
int main(){
    int n,v;
    cin>>n>>v;
    int sum=0,litter=0;
    sum+=min(v,n-1);
    litter=v;
    for(int i=2;i<=n;i++){
        litter--;
        if(litter<v && litter<n-i){
            int temp=v-litter;
            litter+=v-litter;
            sum+=temp*i;
        }
    }
    cout<<sum<<endl;
 }
