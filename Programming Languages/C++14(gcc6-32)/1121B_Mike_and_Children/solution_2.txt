#include<bits/stdc++.h>
using namespace std;
int m[200010];
int main(){
    int n;
    cin>>n;
    int a[n+2];
    for(int i=0;i<n;i++){
        cin>>a[i];
    }
        int cnt=0;
    for(int i=0;i<n;i++){
       for(int j=0;j<i;j++){
        m[a[i]+a[j]]++;
        cnt=max(cnt, m[a[i]+a[j]]);
               }
    }
           cout<<cnt<<endl;
}
