    int j=0,ans=0;
    for(int i=0;i<n;i++){
        cin>>a[i];
     }
     for(int i=0;i<n;i++){
        int temp=a[i];
        mp[temp%k]++;

   // vector<int> vec;
    for(auto it=mp.begin();it!=mp.end();++it){
       if(it->second>1 && it->first==0){
            int temp=it->first;
            //cout<<it->first<<" first loop first "<<it->second<<endl;
            if(it->second%2==0){
                ans+=it->second;
                it->second=-1;
                            }else{
                ans+=it->second-1;
                it->second=1;
            }
                                                      }
    }
    //cout<<ans<<" ans  "<<endl;
    for(auto it=mp.begin();it!=mp.end();++it){
        for(auto jt=mp.begin();jt!=mp.end();++jt){
            if(it->second>0 && jt->second>0 && (it->first+jt->first)%k==0 && jt!=it){
                //cout<<it->first<<" "<<it->second<<endl;
               // cout<<jt->first<<" "<<jt->second<<endl;
                ans+=min(it->second,jt->second)*2;
                if(it->second>jt->second){
                    it->second=it->second-min(it->second,jt->second);
                    jt->second=-1;
                }else{
                    jt->second=jt->second-min(it->second,jt->second);
                    it->second=-1;
                }
                            }
            else if(jt==it && it->second>1 && (it->first*2)%k==0){
                //ans+=(it->second/2)*2;
                if(it->second<jt->second){
                                        ans+=it->second;
                     if(it->second%2==1){
                        ans--;
                    }

                }else{
                    ans+=jt->second;
                    if(jt->second%2==1){
                        ans--;
                    }
                                        jt->second=-1;
                    it->second=it->second-jt->second;
                }
            }
        }
                //cout<<it->first<<" "<<it->second<<endl;
    }
