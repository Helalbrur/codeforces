#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    cin>>n;
    int a[n+2],b[n+2];
    for(int i=0;i<n;i++){
        cin>>a[i]>>b[i];
    }
    int x,cnt=0;
    cin>>x;
    for(int i=0;i<n;i++){
        if(x>a[i] && x>b[i]){
            cnt++;
        }
    }
    cout<<n-cnt<<endl;
 }
