#include<bits/stdc++.h>
using namespace std;
int main(){
    int n;
    //freopen("input.txt","r+",stdin);
    cin>>n;
    int a[n+2];
    for(int i=1;i<=n;i++){
        cin>>a[i];
    }
    uint64_t cnt=0,c=0;
    if(a[n]>0){
        cnt=a[n];
        c=a[n];
    }
    for(int i=n-1;i>=1;i--){
        if(a[i]<c){
            cnt+=a[i];
            c=a[i];
        }else if(a[i]>=c-1 && c>1){
            cnt+=c-1;
            c=c-1;
        }
    }
    cout<<cnt<<endl;
}
