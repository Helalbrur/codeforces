#include<iostream>
#include<string.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    int a[n+2];
    int b[3];
    memset(b,0,sizeof(b));
    for(int i=0;i<n;i++){
        cin>>a[i];
        b[a[i]]++;
    }
    int i=0;
     while(b[1]>0 && b[0]>0 && i<n){
        b[a[i]]--;
        i++;
    }
     cout<<i<<endl;
}
