string sample="ABCDEFGHIJKLMNOPQRSTUVWXYZABCDEFGHIJKLMNOPQRSTUVWXYZABCDEFGHIJKLMNOPQRSTUVWXYZ";
int get(char a,char b){
    int len=sample.length();
    int mn=INT_MAX;
        for(int i=0;i<len;i++){
        int temp=0;
        if(sample[i]==a){
            int j=i;
            while(j>=0){
                if(sample[j]==b){
                    mn=min(temp,mn);
                                   }
                j--;
                temp++;
            }
            temp=0,j=i;
            while(j<len){
                if(sample[j]==b){
                    mn=min(temp,mn);
                                   }
                j++;
                temp++;
            }
        }
    }
    return mn;
}
