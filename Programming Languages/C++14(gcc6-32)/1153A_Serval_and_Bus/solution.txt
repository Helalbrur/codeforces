#include<bits/stdc++.h>
using namespace std;
struct data{
    int value;
    int p;
};
bool cmp(data a,data b){
    if(a.value!=b.value){
        return a.value<b.value;
    }
    return a.p<b.p;
}
int main()
{
    int t,n;
    cin>>n>>t;
    int a[n+5],b[n+5];
    for(int i=1;i<=n;i++){
        cin>>a[i]>>b[i];
    }
        int mn=INT_MAX,pos=-1;
    data d[n+5];
    for(int i=1;i<=n;i++){
        if(a[i]<=t){
            int temp=a[i];
            //cout<<temp<<endl;
            while(temp<=t){
                if(temp==t){
                    break;
                }
                temp+=b[i];
                //cout<<i<<" "<<temp<<endl;
            }
            d[i].p=i;
            d[i].value=temp;
                                 }else {
            d[i].value=a[i];
            d[i].p=i;
        }
    }
    sort(d+1,d+n+1,cmp);
    for(int i=1;i<=n;i++){
       if(d[i].value>=t){
        cout<<d[i].p<<endl;
        return 0;
       }
    }
   }
