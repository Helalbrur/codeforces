 int main()
{
    //freopen("input.txt","r+",stdin);
    int n;
    cin>>n;
    string a;
    cin>>a;
    for(int i=0;i<n-1;i++){
        if(a[i]>a[i+1]){
            cout<<"YES"<<endl;
            cout<<i+1<<" "<<i+2<<endl;
            return 0;
        }
    }
      cout<<"NO"<<endl;
 }
