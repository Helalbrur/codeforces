#include<bits/stdc++.h>
using namespace std;
 map<int,int> mp;
int check(int n)
{
    while(n%10==0){
        n=n/10;
    }
    return n;
}
int f(int n){
    n++;
    while(n%10==0){
        n=n/10;
            }
    return n;
}
int main()
{
    int n;
    cin>>n;
    int cnt=0;
       while(true){
        if(mp[n]>0){
            //cout<<n<<" "<<mp[n]<<endl;
            break;
        }
        mp[n]++;
        cnt++;
        n=f(n);
        n=check(n);
            }
    cout<<cnt<<endl;
}
