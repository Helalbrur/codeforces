{
    int n;

    string s;

    int a[11];
    for(int i=1;i<=9;i++){
        cin>>a[i];
    }
    int start=0,en=0;
    for(int i=0;i<n;i++){
        if((a[s[i]-'0']>(s[i]-'0')) && (start==0 && en==0 )){
            s[i]=a[s[i]-'0']+'0';
            if(start==0){
                start=1;
            }
        }else if((a[s[i]-'0']>=(s[i]-'0')) && (start==1 && en==0)){
            s[i]=a[s[i]-'0']+'0';
                    }
        else if(start==1){
            en=1;
        }
    }
    for(int i=0;i<n;i++){
        cout<<s[i];
    }
    cout<<endl;
}
