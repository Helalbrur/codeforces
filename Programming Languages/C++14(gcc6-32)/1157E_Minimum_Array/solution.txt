#include<bits/stdc++.h>
using namespace std;
int main()
{
        int n;
    cin>>n;
    int a[n+2];
    multiset<int> ms;
    for(int i=0;i<n;i++){
        cin>>a[i];
    }
    int x;
    for(int i=0;i<n;i++){
        cin>>x;
        ms.insert(x);
    }
    for(int i=0;i<n;i++){
        int r=n-a[i];
        if(ms.find(r)!=ms.end()){
            cout<<0<<" ";
            ms.erase(ms.find(r));
            continue;
        }
        auto it=ms.lower_bound(r);
        if(it==ms.end()){
            --it;
        }
        int mx=*it;
        int mn=*ms.begin();
        int rmx=(a[i]+mx)%n;
        int lmx=(a[i]+mn)%n;
        if(rmx<lmx){
            cout<<rmx<<" ";
            ms.erase(it);
        }else{
            cout<<lmx<<" ";
            ms.erase(ms.begin());
        }
            }
    }
