#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n,h,m;
    cin>>n>>h>>m;
    int a[n+3];
    for(int i=1;i<=n;i++){
        a[i]=h;
    }
    for(int i=1;i<=m;i++){
        int l,r,x;
        cin>>l>>r>>x;
        for(int j=l;j<=r;j++){
            a[j]=min(a[j],x);
        }
    }
    long long sum=0;
    for(int i=1;i<=n;i++){
        sum+=a[i]*a[i];
        //cout<<a[i]<<endl;
    }
    cout<<sum<<endl;
}
