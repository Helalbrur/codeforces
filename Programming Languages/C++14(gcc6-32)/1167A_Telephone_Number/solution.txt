#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        string s;
        cin>>s;
        if(n<11){
            cout<<"NO"<<endl;
        }else{
            int l=n-10;
            bool ok=false;
            for(int i=0;i<l;i++){
                if(s[i]=='8'){
                    ok=true;
                    break;
                }
            }
            if(ok==true){
                cout<<"YES"<<endl;
            }else{
                cout<<"NO"<<endl;
            }
        }
    }
}
