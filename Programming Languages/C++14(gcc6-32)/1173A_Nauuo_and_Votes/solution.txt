#include<iostream>
#include<math.h>
using namespace std;
int main()
{
    int u,d,z;
    cin>>u>>d>>z;
    if(z==0){
        if(u>d){
            cout<<"+"<<endl;
        }else if(d>u){
            cout<<"-"<<endl;
        }else{
            cout<<0<<endl;
        }
    }else{
        if(u>d+z){
            cout<<"+"<<endl;
        }else if(d>u+z){
            cout<<"-"<<endl;
        }else{
            cout<<"?"<<endl;
        }
    }
}
