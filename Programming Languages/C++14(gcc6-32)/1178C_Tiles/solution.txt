#include<bits/stdc++.h>
using namespace std;
long long bigMod ( long long a, int p, int m )
{
    if ( p == 0 )return 1; 
     if ( p & 1 ) 
    {
        return ( ( a % m ) * ( bigMod ( a, p - 1, m ) ) ) % m;
    }
    else
    {
        long long tmp = bigMod ( a, p / 2, m ); 
        return ( (tmp%m) * (tmp%m )) % m;
    }
}
