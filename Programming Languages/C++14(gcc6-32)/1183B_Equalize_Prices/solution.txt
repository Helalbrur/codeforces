#include<bits/stdc++.h>
using namespace std;
 int main(){
   int q;
   cin>>q;
   for(int t=1;t<=q;t++){
        int n,k;
        cin>>n>>k;
        int a[n+2];
        for(int i=0;i<n;i++){
            cin>>a[i];
        }
        int flag=0,pos;
        sort(a,a+n);
        a[n]=a[0]+k;
        for(int i=n;i>=0;i--){
            if(flag==1){
                break;
            }
            int s=0;
            for(int j=0;j<=n;j++){
                if(abs(a[i]-a[j])>k){
                    s++;
                }
            }
            if(s==0){
                pos=a[i];
                flag=1;
                break;
            }
        }
        if(flag==0){
            cout<<"-1"<<endl;
        }else{
            cout<<pos<<endl;
        }
   }
}
