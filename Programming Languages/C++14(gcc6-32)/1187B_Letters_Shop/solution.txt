#include<bits/stdc++.h>
using namespace std;
int b[27][200010];
int main()
{
    int n;
    cin>>n;
    string s;
    cin>>s;
    int m;
    cin>>m;
        int a[27];
    memset(a,0,sizeof(a));
    memset(b,0,sizeof(b));
    for(int i=0;i<n;i++){
        a[s[i]-'a']++;
        b[s[i]-'a'][a[s[i]-'a']]=i;
    }
        while(m--){
        string text;
        cin>>text;
        int x[27];
        memset(x,0,sizeof(x));
        int cnt=0;
        for(int i=0;i<text.length();i++){
            x[text[i]-'a']++;
            cnt=max(cnt,b[text[i]-'a'][x[text[i]-'a']]);
        }
        cout<<cnt+1<<endl;
    }
}
