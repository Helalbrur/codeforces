#include<bits/stdc++.h>
using namespace std;
int main()
{
    //freopen("a.txt","r+",stdin);
    int t;
    cin>>t;
    while(t--){
        int b,p,f;
        cin>>b>>p>>f;
        int h,c;
        cin>>h>>c;
        int hm=0;
        int cc=0;
        if(h>c){
            hm=min((b/2),p)*h;
            b=b-(min((b/2),p))*2;
            cc=min((b/2),f)*c;
        }else{
            cc=(min((b/2),f)*c);
            b=b-(min((b/2),f))*2;
            hm=min((b/2),p)*h;
        }
        cout<<hm+cc<<endl;
    }
}
