#include<bits/stdc++.h>
using namespace std;
int main()
{
    //freopen("b.txt","r+",stdin);
    int n,m;
    cin>>n>>m;
    int a[n+5][m+5];
    vector<pair<int,int>> v;
    int sum_a=0,sum_b=0;
    for(int i=1;i<=n;i++){
        for(int j=1;j<=m;j++){
            cin>>a[i][j];
            sum_a+=a[i][j];
        }
    }
    int b[n+2][m+2];
    memset(b,0,sizeof(b));
    for(int i=1;i<n;i++){
        for(int j=1;j<m;j++){
             if(a[i][j]==1 && a[i][j+1]==1 && a[i+1][j]==1 && a[i+1][j+1]==1){
                 v.push_back(make_pair(i,j));
                b[i][j]=1;
                b[i][j+1]=1;
                b[i+1][j]=1;
                b[i+1][j+1]=1;
            }
         }
    }
    for(int i=1;i<=n;i++){
        for(int j=1;j<=m;j++){
            sum_b+=b[i][j];
        }
    }
     int t=v.size();
    if(sum_a==sum_b){
        cout<<t<<endl;
         for(int i=0;i<t;i++){
            cout<<v[i].first<<" "<<v[i].second<<endl;
        }
    }
    else{
        cout<<"-1"<<endl;
    }
 }
