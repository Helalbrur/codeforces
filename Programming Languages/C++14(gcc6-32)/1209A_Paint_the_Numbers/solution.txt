#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    int a[n+3];
    int vis[107];
    for(int i=0;i<n;i++){
        cin>>a[i];
    }
    sort(a,a+n);
    memset(vis,0,sizeof(vis));
    int cnt=0;
    for(int i=0;i<n;i++){
        if(vis[a[i]]==0){
            cnt++;
            for(int j=i+1;j<n;j++){
                if(a[j]%a[i]==0){
                    vis[a[j]]=1;
                }
            }
        }
    }
    cout<<cnt<<endl;
}
