#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    string s;
    cin>>s;
    if(n%2){
        cout<<"No"<<endl;
        return 0;
    }
    stack<char> q;
    int op=0,cl=0,c=0;
     for(int i=0;i<s.length();i++){
        if(s[i]=='('){
            q.push('(');
            op++;
        }else{
            if(q.size()>0){
                q.pop();
             }else{
                c++;
            }
            cl++;
        }
    }
    if(op!=cl){
        cout<<"No"<<endl;
    }
    else if(q.empty()){
        cout<<"Yes"<<endl;
    }else if(c==1 && q.top()=='('){
        cout<<"Yes"<<endl;
    }else{
        cout<<"No"<<endl;
    }
}
