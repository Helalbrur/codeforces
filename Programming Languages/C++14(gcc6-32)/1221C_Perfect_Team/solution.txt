#include<bits/stdc++.h>
using namespace std;
int main()
{
    int q;
    cin>>q;
    while(q--){
        long long c,m,n,s=0;
        cin>>c>>m>>n;
        long long x=min(m,c);
        s=c+m+n;
        cout<<min(x,s/3)<<endl;
    }
}
