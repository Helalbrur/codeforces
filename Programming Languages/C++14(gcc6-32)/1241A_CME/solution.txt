#include<bits/stdc++.h>
using namespace std;
int main()
{
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    int q;
    cin>>q;
    while(q--){
        int n;
        cin>>n;
        if(n==2){
            cout<<2<<endl;
        }else if(n%2==0){
            cout<<0<<endl;
        }else{
            cout<<1<<endl;
        }
    }
}
