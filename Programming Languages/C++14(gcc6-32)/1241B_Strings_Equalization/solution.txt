#include<bits/stdc++.h>
using namespace std;
int main()
{
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    int q;
    cin>>q;
    while(q--)
    {
        string s,t;
        cin>>s;
        cin>>t;
        int a[27],b[27];
        memset(a,0,sizeof(a));
        memset(b,0,sizeof(b));
        for(int i=0;i<s.length();i++){
            a[s[i]-'a']++;
            b[t[i]-'a']++;
        }
        bool ok=false;
        for(int i=0;i<26;i++){
            if(a[i]>0 && b[i]>0){
                ok=true;
            }
        }
        if(ok==true){
            cout<<"YES"<<endl;
        }
        else{
            cout<<"NO"<<endl;
        }
    }
}
