#include<bits/stdc++.h>
using namespace std;
 int main()
{
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        string a;
        cin>>a;
        int i=0,f=0;
         while(i<n && a[i]!='1'){
             f++;
            i++;
        }
        int b=0;
        i=n-1;
        while(i>=0 && a[i]!='1'){
            b++;
            i--;
        }
       cout<<max(n,2*n-min(b,f)*2)<<endl;
     }
}
