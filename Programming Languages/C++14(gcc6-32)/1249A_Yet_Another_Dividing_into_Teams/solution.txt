#include<bits/stdc++.h>
using namespace std;
int main()
{
    int q;
    cin>>q;
    while(q--){
        int n;
        cin>>n;
        int a[n+2];
        for(int i=0;i<n;i++){
            cin>>a[i];
        }
        sort(a,a+n);
        bool ok=false;
        for(int i=1;i<n;i++){
            if(abs(a[i]-a[i-1])==1){
                ok=true;
                break;
            }
        }
        if(ok==true){
            cout<<2<<endl;
         }else{
            cout<<1<<endl;
        }
    }
}
