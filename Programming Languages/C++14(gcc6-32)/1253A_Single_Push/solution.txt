#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        int cnt=0;
        int a[n+2],b[n+2];
        for(int i=1;i<=n;i++){
            cin>>a[i];
        }
        int f=-1,l=-1;
        bool ok=true;
        for(int i=1;i<=n;i++){
            cin>>b[i];
            if(a[i]>b[i]){
                ok=false;
            }
            if(a[i]!=b[i]){
                cnt++;
                if(f==-1){
                    f=i;
                }
                else{
                    l=i;
                }
            }
        }
        if(ok==false){
            cout<<"NO"<<endl;
        }
        else if(cnt<=1){
            cout<<"YES"<<endl;
        }else{
            int dif=b[f]-a[f];
            for(int i=f;i<=l;i++){
                if(b[i]!=(a[i]+dif)){
                    ok=false;
                    break;
                }
            }
             if(ok==false){
                cout<<"NO"<<endl;
            }else{
                cout<<"YES"<<endl;
            }
        }
    }
}
