#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--){
        int a,b;
        cin>>a>>b;
        if(a==b){
            cout<<0<<endl;
        }else{
          int dif=abs(a-b);
          int cnt=0;
          if(dif>=5){
            cnt+=dif/5;
            dif=dif-cnt*5;
          }
          if(dif>=2){
            int c=dif/2;
            cnt+=c;
            dif=dif-c*2;
          }
          if(dif){
            cnt++;
          }
          cout<<cnt<<endl;
         }
    }
}
