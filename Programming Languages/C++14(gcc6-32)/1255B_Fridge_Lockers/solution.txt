#include<bits/stdc++.h>
using namespace std;
struct st{
    int val;
    int id;
};
bool cmp(st a,st b){
    return a.val<b.val;
}
int main()
{
    int t;
    cin>>t;
    while(t--){
         int n,m;
        cin>>n>>m;
        st a[n+2];
        int sum=0;
        for(int i=0;i<n;i++){
             cin>>a[i].val;
            a[i].id=i+1;
        }
        if(n>m || n==2){
            cout<<-1<<endl;
        }else if(m==n){
            sort(a,a+n,cmp);
            vector<pair<int,int>> v;
            for(int i=0;i<n;i++){
                if(i==0){
                     sum+=a[i].val;
                    sum+=a[n-1].val;
                    v.push_back(make_pair(a[i].id,a[n-1].id));
                 }else{
                    sum+=a[i].val;
                    sum+=a[i-1].val;
                    v.push_back(make_pair(a[i].id,a[i-1].id));
                }
            }
            cout<<sum<<endl;
             for(int i=0;i<v.size();i++){
                cout<<v[i].first<<" "<<v[i].second<<endl;
             }
         }
      }
}
