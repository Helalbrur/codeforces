#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        int a[n+2];
        memset(a,-1,sizeof(a));
        int mx=INT_MAX;
        for(int i=0;i<n;i++){
            int x;
            cin>>x;
            if(a[x]!=-1){
                int l=abs(i-a[x])+1;
                mx=min(mx,l);
            }
            a[x]=i;
        }
        if(mx==INT_MAX){
            cout<<"-1"<<endl;
        }else{
            cout<<mx<<endl;
        }
     }
}
