#include<bits/stdc++.h>
using namespace std;
int main(){
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        int a[n+2];
        int d[n+2];
        string b;
        int mx=0,mn=n+1;
        for(int i=1;i<=n;i++){
            cin>>a[i];
            d[a[i]]=i;
         }
        for(int i=1;i<=n;i++){
            mx=max(mx,d[i]);
            mn=min(mn,d[i]);
            if(mx-mn<i){
                b+='1';
            }else{
                b+='0';
            }
         }
        cout<<b<<endl;
    }
}
