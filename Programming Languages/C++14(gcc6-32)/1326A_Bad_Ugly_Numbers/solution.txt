#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        string s;
        int sum=2;
        s+='2';
        for(int i=2;i<=n;i++){
            s+='3';
            sum+=3;
        }
        if(sum%3==0 || (s[n-1]-'0')%2==0){
            s="-1";
        }
        cout<<s<<"\n";
    }
}
