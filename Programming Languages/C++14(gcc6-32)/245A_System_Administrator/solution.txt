#include<iostream>
#include<string.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    int a[3][3];
    memset(a,0,sizeof(a));
    while(n--){
        int t,x,y;
        cin>>t>>x>>y;
        a[t][0]+=x;
        a[t][1]+=y;
    }
    if(a[1][0]>=a[1][1]){
        cout<<"LIVE"<<endl;
    }else{
        cout<<"DEAD"<<endl;
    }
    if(a[2][0]>=a[2][1]){
        cout<<"LIVE"<<endl;
    }else{
        cout<<"DEAD"<<endl;
    }
}
