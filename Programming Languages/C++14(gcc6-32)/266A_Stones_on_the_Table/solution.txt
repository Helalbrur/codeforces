#include<iostream>
#include<string>
using namespace std;
int main()
{
    int n;
    cin>>n;
    string a;
    cin>>a;
    int h=n,i=1;
    while(i<h){
        if(a[i]==a[i-1]){
            a.erase(i,1);
            h--;
            i--;
        }
        i++;
    }
    cout<<n-a.length()<<endl;
}
