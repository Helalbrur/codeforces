#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    int a[1010];
    memset(a,0,sizeof a);
    for(int i=0;i<n;i++)
    {
        int x;
        cin>>x;
        a[x]++;
    }
    for(int i=1;i<=1000;i++)
    {
        if(a[i]>((double)n/2+.5))
        {
           //cout<<a[i]<<endl;
            cout<<"NO"<<endl;
            return 0;
        }
    }
    cout<<"YES"<<endl;
}
