#include<bits/stdc++.h>
using namespace std;
int main()
{
    string s;
    //freopen("input.txt","r+",stdin);
    cin>>s;
    int n,x,mx=0;
    cin>>n;
    map<char,int> mp;
    for(char a='a' ;a<='z';a++)
    {
        cin>>x;
        mp[a]=x;
        mx=max(mx,x);
    }
    int sum=0,l=s.length();
    for(int i=0;i<l;i++)
    {
        sum+=mp[s[i]]*(i+1);
    }
    for(int i=l+1;i<=l+n;i++)
    {
        sum+=mx*i;
    }
    cout<<sum<<endl;
}
