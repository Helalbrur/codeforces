#include<bits/stdc++.h>
using namespace std;
int main()
{
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    //freopen("input.txt","r",stdin);
    int n;
    cin>>n;
    int a[n+5],cnt=0,temp=-1;
    for(int i=0;i<n;i++)
    {
        cin>>a[i];
        if(i>0)
        {
            if(a[i]<a[i-1])
            {
                cnt++;
                temp=n-i;
            }
         }
    }
    if(cnt==0)
    {
        cout<<0<<endl;
        return 0;
    }
    else if(cnt>1)
    {
        cout<<"-1"<<endl;
        return 0;
    }
    else
    {
        if(a[n-1]<=a[0])
            cout<<temp<<endl;
        else
            cout<<"-1"<<endl;
    }
}
