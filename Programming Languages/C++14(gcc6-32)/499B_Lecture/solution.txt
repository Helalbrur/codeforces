#include<bits/stdc++.h>
using namespace std;
int main()
{
    ios_base::sync_with_stdio(0);
    cin.tie(NULL);
    //freopen("input.txt","r+",stdin);
    int n,k;
    cin>>n>>k;
    string f,s;
    map<string,string> mp;
    for(int i=0;i<k;i++)
    {
        cin>>f>>s;
        if(f.length()>s.length())
        {
            mp[f]=s;
            mp[s]=s;
        }
        else
        {
            mp[f]=f;
            mp[s]=f;
        }
    }
    for(int i=0;i<n;i++)
    {
        cin>>s;
        if(i>0)
            cout<<" ";
        cout<<mp[s];
    }
    cout<<endl;
}
