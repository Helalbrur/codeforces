#include<iostream>
using namespace std;
long long int bigMod(long long int a,long long int p,long long int m){
    if(p==0){
        return a%m;
    }
    long long int x=bigMod(a,p/2,m);
    x=((x%m)*(x%m))%m;
    if(p%2){
        x=((x%m)*(a%m))%m;
    }
    return x;
}
int main()
{
    long long int n;
    cin>>n;
    cout<<bigMod(5,n,100)<<endl;
}
