#include<iostream>
using namespace std;
struct data{
    int score;
    int time;
};
int main()
{
    int n,c;
    cin>>n>>c;
    data a[n+3];
    for(int i=1;i<=n;i++){
        cin>>a[i].score;
    }
    for(int i=1;i<=n;i++){
        cin>>a[i].time;
    }
    int Limak=0,Radewoosh=0;
    int t=0;
    for(int i=1;i<=n;i++){
        t+=a[i].time;
        Limak+=max(a[i].score-t*c,0);
    }
    t=0;
    for(int i=n;i>=1;i--){
        t+=a[i].time;
        Radewoosh+=max(a[i].score-t*c,0);
    }
    if(Limak>Radewoosh){
        cout<<"Limak"<<endl;
    }else if(Limak<Radewoosh){
        cout<<"Radewoosh"<<endl;
    }else{
        cout<<"Tie"<<endl;
    }
}
