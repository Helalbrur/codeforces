#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    string a[n+3];
    bool ok=false;
    for(int i=0;i<n;i++){
        cin>>a[i];
        if(ok==false){
            if(a[i][0]=='O' && a[i][1]=='O'){
                a[i][0]='+';
                a[i][1]='+';
                ok=true;
            }else if(a[i][3]=='O' && a[i][4]=='O'){
                 a[i][3]='+';
                 a[i][4]='+';
                 ok=true;
            }
        }
    }
    if(ok==true){
        cout<<"YES"<<endl;
        for(int i=0;i<n;i++){
            cout<<a[i]<<endl;
        }
    }else{
        cout<<"NO"<<endl;
    }
}
