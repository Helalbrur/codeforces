#include<iostream>
using namespace std;
int main()
{
    int n,c;
    cin>>n>>c;
    int a[n+2];
    int cnt=0;
    for(int i=1;i<=n;i++){
        cin>>a[i];
        cnt++;
        if(i>1){
            if(a[i]-a[i-1]>c){
                cnt=1;
            }
        }
    }
    cout<<cnt<<endl;
}
