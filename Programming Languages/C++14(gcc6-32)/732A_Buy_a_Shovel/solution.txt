#include<iostream>
using namespace std;
int main()
{
    int k,r;
    cin>>k>>r;
    for(int i=1;i<=10000000;i++){
        if((k*i)%10==0 or (k*i > r && (k*i-r)%10==0) or (k*i==r)){
            cout<<i<<endl;
            return 0;
        }
    }
}
