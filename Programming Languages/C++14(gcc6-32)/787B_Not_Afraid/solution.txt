#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n,m;
    //freopen("input.txt","r+",stdin);
    cin>>n>>m;
    int flag[m+5];
    memset(flag,0,sizeof flag);
    for(int i=1;i<=m;i++)
    {
        int k;
        cin>>k;
        int a;
        map<int,int>mp;
        for(int j=1;j<=k;j++)
        {
            cin>>a;
            mp[a]++;
            if(mp.find(-a)==mp.end())
            {
                flag[i]=flag[i];
            }
            else
            {
                flag[i]=1;
            }
        }
    }
    for(int i=1;i<=m;i++)
    {
        if(flag[i]==0)
        {
            cout<<"YES"<<endl;
            return 0;
        }
    }
    cout<<"NO"<<endl;
}
