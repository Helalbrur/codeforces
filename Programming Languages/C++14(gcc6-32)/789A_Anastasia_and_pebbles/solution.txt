#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n,k;
    cin>>n>>k;
    int a;
    queue<int> q;
    int d=0,cnt=0;
    for(int i=0;i<n;i++)
    {
        cin>>a;
        d+=a/k;
        if(a%k!=0)
        q.push(a-(k*(a/k)));
    }
    while(!q.empty())
    {
        a=q.front();
        q.pop();
        d++;
        if(a>k)
            q.push(a-k);
    }
    if(d%2==0)
        cout<<d/2<<endl;
    else
        cout<<d/2+1<<endl;
}
