#include<bits/stdc++.h>
using namespace std;
int main()
{
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    //freopen("input.txt","r",stdin);
    int n,k;
     cin>>n>>k;
    int a[n+5];
    for(int i=0;i<n;i++)
    {
        cin>>a[i];
    }
    sort(a,a+n);
    long long int s=0;
    for(int i=0;i<k;i++)
    {
        s+=(long long int) a[i];
    }
    cout<<s<<endl;
}
