#include<bits/stdc++.h>
using namespace std;
//typedef unsigned long long ll;
int main()
{
    int n,cnt=0;
    cin>>n;
    for(int i=1;i<n;i++)
    {
        cnt+=(i+i+1)%(n+1);
        cnt=cnt%n;
    }
    cout<<cnt<<endl;
}
