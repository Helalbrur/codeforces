#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n,a,b,x,y;
    cin>>n>>a>>b>>x>>y;
    int p=(n*a)+x*2;
    int q=(n*b)+2*y;
    if(p>q){
        cout<<"Second"<<endl;
    }else if(q>p){
        cout<<"First"<<endl;
    }else{
        cout<<"Friendship"<<endl;
    }
}
