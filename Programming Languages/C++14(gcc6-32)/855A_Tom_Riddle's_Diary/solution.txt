#include<iostream>
#include<map>
using namespace std;
int main()
{
    int n;
    cin>>n;
    map<string,int> mp;
    while(n--){
        string a;
        cin>>a;
        if(mp.find(a)==mp.end()){
            cout<<"NO"<<endl;
        }else{
            cout<<"YES"<<endl;
        }
        mp[a]++;
    }
}
