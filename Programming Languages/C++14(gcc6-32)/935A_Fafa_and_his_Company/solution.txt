#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    int sq=sqrt(n);
    int cnt=1;
    for(int i=2;i<=sq;i++){
        if(n%i==0){
            if(n/i==i){
                cnt+=1;
            }else{
                cnt+=2;
            }
         }
    }
    cout<<cnt<<endl;
}
