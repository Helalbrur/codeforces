#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    string a;
    cin>>a;
    int cnt=0;
    for(int i=0;i<n;i++){
        int x=0;
        while(a[i]=='x'){
            x++;
            i++;
        }
        if(x>=3){
            cnt+=x-2;
        }
     }
    cout<<cnt<<endl;
}
