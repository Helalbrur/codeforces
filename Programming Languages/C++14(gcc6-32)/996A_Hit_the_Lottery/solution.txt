#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    int cnt=0;
    while(n>=100){
        int c=n/100;
        cnt+=c;
        n=n-c*100;
    }
    while(n>=20){
        int c=n/20;
        cnt+=c;
        n=n-c*20;
    }
     while(n>=10){
        int c=n/10;
        cnt+=c;
        n=n-c*10;
    }
     while(n>=5){
        int c=n/5;
        cnt+=c;
        n=n-c*5;
    }
     while(n>=1){
        int c=n/1;
        cnt+=c;
        n=n-c*1;
    }
    cout<<cnt<<endl;
 }
