#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        int a[n+3];
        a[0]=0;
        bool ok=false;
        for(int i=1;i<=n;i++){
            cin>>a[i];
            if(i>1 && abs(a[i]-a[i-1])>1 && ok==false){
                cout<<"YES"<<endl;
                cout<<i-1<<" "<<i<<endl;
                ok=true;
            }
        }
         if(ok==false){
            cout<<"NO"<<endl;
        }
    }
}
