#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        int a[n+3];
        int64_t s=0,o=0;
        for(int i=0;i<n;i++){
            cin>>a[i];
            s+=a[i];
            o^=a[i];
        }
        if(s==o*2){
            cout<<0<<endl;
            cout<<endl;
        }else{
            s+=o;
            cout<<2<<endl;
            cout<<o <<" "<<s<<endl;
        }
     }
}
