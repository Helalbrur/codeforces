#include<bits/stdc++.h>
    using namespace std;
    int64_t gcd(int64_t a, int64_t b){
        if (a == 0)
            return b;
        return gcd(b % a, a);
     }
    int64_t lcm(int64_t a, int64_t b)
     {
        return (a*b)/gcd(a, b);
     }
    int main()
    {
        int64_t n;
        cin>>n;
         int64_t p=n,q=n;
        for(int64_t i=1;i*i<=n;i++){
            if(n%i==0){
                int64_t u=n/i;
                if(lcm(u,i)==n){
                    if(max(p,q)>max(u,i)){
                        p=min(u,i);
                        q=max(u,i);
                    }
                 }
            }
          }
        cout<<p<<" "<<q<<endl;
    }
