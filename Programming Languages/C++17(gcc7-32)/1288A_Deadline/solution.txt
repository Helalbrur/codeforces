#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--){
        int n,d;
        cin>>n>>d;
        if(d<=n){
            cout<<"YES"<<endl;
        }else{
             bool ok=false;
            for(int i=1;i<n;i++){
                double x=ceil(double(d)/double(i+1));
                if((int)x+i<=n){
                    ok=true;
                    break;
                }
            }
            if(ok==true){
                cout<<"YES"<<endl;
            }else{
                cout<<"NO"<<endl;
            }
        }
    }
}
