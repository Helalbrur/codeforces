#include<bits/stdc++.h>
using namespace std;
long long int f(long long int b){
    long long int c=0;
    while(b){
        c++;
        b=b/10;
     }
    return c-1;
}
int main()
{
    int t;
    cin>>t;
    while(t--){
         long long int a,b;
        cin>>a>>b;
        cout<<a*f(b+1)<<endl;
    }
}
