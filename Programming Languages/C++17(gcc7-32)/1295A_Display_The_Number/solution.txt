#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        string s;
        if(n%2){
            s='7';
            n-=3;
         }
        n=n/2;
        for(int i=1;i<=n;i++){
            s+='1';
        }
        cout<<s<<endl;
    }
}
