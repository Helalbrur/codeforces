#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--){
        int n,m;
        cin>>n>>m;
        int a[n+2],b[n+2];
        memset(b,0,sizeof(b));
        for(int i=1;i<=n;i++){
            cin>>a[i];
        }
        for(int i=1;i<=m;i++){
            int x;
            cin>>x;
            b[x]=1;
        }
        bool ok=true;
        for(int j=1;j<=n;j++){
            for(int i=1;i<n;i++){
                if(a[i]>a[i+1]){
                    if(b[i]==0){
                        ok=false;
                        break;
                    }else{
                        swap(a[i],a[i+1]);
                    }
                }
            }
        }
        if(ok){
            cout<<"Yes\n";
        }else{
            cout<<"No\n";
        }
    }
}
