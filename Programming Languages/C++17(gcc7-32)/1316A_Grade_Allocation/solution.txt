#include<bits/stdc++.h>
using namespace std;
int main()
{
    int t;
    cin>>t;
    while(t--){
        long long int n,m;
        cin>>n>>m;
        int a[n+2];
        long long int s=0;
        for(int i=0;i<n;i++){
            cin>>a[i];
            s+=a[i];
        }
        cout<<min(s,m)<<endl;
    }
}
