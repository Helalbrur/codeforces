#include<bits/stdc++.h>
using namespace std;
int main(){
     int n,m;
    //freopen("6a.txt","r+",stdin);
    cin>>n>>m;
    int cnt=0;
    int mx=max(n,m);
    int mn=min(n,m);
    while(mx>=2 && mn>=1)
    {
        int a=mx;
        int b=mn;
        a=a-2;
        b=b-1;
        cnt=cnt+1;
        //cout<<mx<<" "<<mn<<" "<<cnt<<endl;
        mx=max(a,b);
        mn=min(a,b);
     }
    cout<<cnt<<endl;
}
