#include<bits/stdc++.h>
using namespace std;
int main()
{
     int n,m,k;
    //freopen("123.txt","r",stdin);
    cin>>n>>m>>k;
    int a[n+5];
    int mn=1000;
    for(int i=1;i<=n;i++)
    {
        cin>>a[i];
        if(a[i]&& a[i]<=k && mn>(abs(m-i)*10) )
        {
            mn=abs(m-i)*10;
        }
    }
    cout<<mn<<endl;
}
