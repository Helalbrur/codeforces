#include<bits/stdc++.h>
using namespace std;
int main()
{
    int n;
    cin>>n;
    int b[n+5];
    int sum=0;
    for(int i=0;i<n;i++)
    {
        cin>>b[i];
        if(b[i]>0)
            sum+=b[i];
    }
    if(sum%2==1)
    {
        cout<<sum<<endl;
        return 0;
    }
    int mx=-9999999999;
    for(int i=0;i<n;i++)
    {
       if(b[i]>=0)
       {
           if(((sum-b[i])%2==1)&& ( (sum-b[i])>mx))
           {
               mx=sum-b[i];
           }
        }
       else{
        if((abs(sum+b[i])%2==1)&& ( sum+b[i]>mx))
           {
               mx=sum+b[i];
           }
       }
    }
    cout<<mx<<endl;
}
