#include<bits/stdc++.h>
using namespace std;
int main()
{
    string a,b,c;
    cin>>a;
    cin>>b;
    int l=a.length();
    for(int i=0;i<l;i++)
    {
        if(a[i]<b[i])
        {
            cout<<"-1"<<endl;
            return 0;
        }
    }
    for(int i=0;i<l;i++)
    {
         if(a[i]==b[i])
        {
           if(a[i]=='z')
           {
               cout<<a[i];
           }
           else
            printf("%c",b[i]+1);
        }
        else
        {
            cout<<b[i];
        }
    }
    cout<<endl;
}
