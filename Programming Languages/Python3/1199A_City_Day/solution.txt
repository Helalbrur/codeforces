n,x,y=map(int,input().split())
  a = [int(x) for x in input().split()]
for i in range(0,n):
    flag=0
    if i-x>=0:
        p=i-x
    else:
        p=0;
    if i+y>=n:
        q=n
    else:
        q=i+y+1
    for j in range(i,p,-1):
        if a[j]<a[i]:
            flag=1
    for j in range(i,q):
        if a[j]<a[i]:
            flag=1
    if flag==0:
        print(i+1)
        break
