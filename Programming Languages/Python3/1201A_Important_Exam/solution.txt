    s[i]=input()
b=list(map(int,input().split()))
 for i in range(n):
    for j in range(q):
        t=ord(s[i][j])-ord('A');
        a[j][t]=a[j][t]+1
        #print(s[i][j],t,a[j][t])
        cnt=0
#print(a)
for i in range(q):
